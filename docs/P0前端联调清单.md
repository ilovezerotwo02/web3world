## 通用工具函数

```typescript
// 基础地址：本地开发为 http://localhost:3000，部署后为生产 URL
const API_BASE = "http://localhost:3000";

// 从 localStorage 读取 token
function getToken(): string | null {
  return localStorage.getItem("auth_token");
}

// 通用请求头
function headers(authenticated = false): Record<string, string> {
  const h: Record<string, string> = { "Content-Type": "application/json" };
  if (authenticated) {
    const token = getToken();
    if (token) h["Authorization"] = `Bearer ${token}`;
  }
  return h;
}

// 统一错误处理：401 时跳转登录页
async function handleResponse(res: Response): Promise<any> {
  if (res.status === 401) {
    localStorage.removeItem("auth_token");
    window.location.href = "/login";
    throw new Error("未认证，请重新登录");
  }
  if (res.status === 403) {
    throw new Error("权限不足");
  }
  if (res.status === 503) {
    throw new Error("系统维护中，请稍后重试");
  }
  const data = await res.json();
  if (!res.ok) {
    throw new Error(data.error || `请求失败 (${res.status})`);
  }
  return data;
}
```

## 详细接口说明 + 调用示例

### 1. 用户注册/登录

```typescript
// 注册新用户
async function register(username: string, avatarUrl?: string) {
  const res = await fetch(`${API_BASE}/api/auth`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ username, avatarUrl }),
  });
  const data = await handleResponse(res);

  // 保存 token 到 localStorage
  localStorage.setItem("auth_token", data.token);

  return data; // { user: { id, username, role, exp, level }, token: "..." }
}

// 获取当前登录用户信息
async function getCurrentUser() {
  const res = await fetch(`${API_BASE}/api/auth`, {
    headers: headers(true),
  });
  return handleResponse(res); // { user: { id, username, role, exp, level } }
}

// 退出登录
async function logout() {
  const res = await fetch(`${API_BASE}/api/auth`, {
    method: "DELETE",
    headers: headers(true),
  });
  const data = await handleResponse(res);
  localStorage.removeItem("auth_token");
  return data; // { message: "已退出登录" }
}
```

**Response 示例：**
```json
// POST /api/auth 成功 (201)
{
  "user": { "id": "uuid", "username": "test_user", "role": "user", "exp": 0, "level": 1 },
  "token": "session-token-string"
}

// 错误：用户名已存在 (409)
{ "error": "用户名已存在" }

// 错误：用户名为空 (400)
{ "error": "用户名不能为空" }
```

- 返回的 `token` 存入 `localStorage`，key 为 `auth_token`
- 后续所有需认证的请求在 `Authorization` header 中携带
- 用户名 1-20 字符，超长返回 400

### 2. 认证鉴权

```typescript
// 封装一个通用 fetch，自动携带 token 并处理 401
async function apiFetch(path: string, options: RequestInit = {}) {
  const token = getToken();
  const res = await fetch(`${API_BASE}${path}`, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...(options.headers || {}),
    },
  });

  // 401 → 清除 token 并跳转登录页
  if (res.status === 401) {
    localStorage.removeItem("auth_token");
    window.location.href = "/login";
    throw new Error("未认证");
  }

  // 403 → 权限不足提示
  if (res.status === 403) {
    throw new Error("权限不足");
  }

  // 503 → 维护模式提示
  if (res.status === 503) {
    const body = await res.json().catch(() => ({}));
    throw new Error(body.error || "系统维护中，请稍后重试");
  }

  const data = await res.json();
  if (!res.ok) {
    throw new Error(data.error || `请求失败 (${res.status})`);
  }
  return data;
}

// 使用示例
const user = await apiFetch("/api/level"); // 自动携带 token
```

**前端全局拦截逻辑：**
- 所有 API 请求通过 `apiFetch` 封装
- 401 响应 → 清除 `auth_token` → 跳转 `/login`
- 403 响应 → 弹窗提示"权限不足"
- 503 响应 → 显示维护中页面

### 4. 进度同步

```typescript
// 获取课程进度
async function getProgress(userId: string, courseId: string) {
  const res = await fetch(
    `${API_BASE}/api/progress?userId=${userId}&courseId=${courseId}`,
    { headers: headers() },
  );
  const data = await handleResponse(res);
  return data.progress; // FarmProgress | null
}

// 应用进度动作（状态机驱动）
async function applyProgressAction(
  userId: string,
  courseId: string,
  action: { type: string; chapter?: number; answers?: any },
) {
  const res = await fetch(`${API_BASE}/api/progress`, {
    method: "POST",
    headers: headers(),
    body: JSON.stringify({ userId, courseId, action }),
  });
  return handleResponse(res);
  // 返回: { progress: FarmProgress, changed: boolean }
}

// 使用示例：用户完成第 1 章学习
async function onChapterComplete(userId: string, courseId: string, chapterIndex: number) {
  const result = await applyProgressAction(userId, courseId, {
    type: "complete-theory",
    chapter: chapterIndex,
  });
  if (result.changed) {
    console.log("进度已更新:", result.progress);
  }
}

// 重置进度
async function resetProgress(userId: string, courseId: string) {
  const res = await fetch(
    `${API_BASE}/api/progress?userId=${userId}&courseId=${courseId}`,
    { method: "DELETE", headers: headers() },
  );
  return handleResponse(res);
  // 返回: { deleted: true }
}
```

**action 类型与前端事件映射：**

```typescript
async function bindWallet(address: string) {

async function getWalletByUser(userId: string) {
  const res = await fetch(`${API_BASE}/api/wallet?userId=${userId}`, {
    headers: headers(),
  });
  return handleResponse(res);
  // 返回: { user: { id, username, walletAddress } | null }
}

async function getWalletByAddress(address: string) {
  const res = await fetch(`${API_BASE}/api/wallet?address=${address}`, {
    headers: headers(),
  });
  return handleResponse(res);
}

// 前端连接 MetaMask 后调用示例
async function connectMetaMask() {
  if (!window.ethereum) {
    alert("请安装 MetaMask");
    return;
  }
  const accounts = await window.ethereum.request({ method: "eth_requestAccounts" });
  const address = accounts[0].toLowerCase();
  try {
    const result = await bindWallet(address);
    console.log("钱包绑定成功:", result.user.walletAddress);
  } catch (err) {
    if (err.message.includes("已被绑定")) {
      alert("该钱包地址已被其他账号绑定");
    }
  }
}
```

**注意：**
- `address` 必须为 `0x` 开头的 42 字符以太坊地址（小写）
- 首次绑定自动发放 15 EXP，后续修改不再发放（幂等）
- 地址已被其他用户绑定时返回 409

### 6. 断点恢复

```typescript
// 保存进度断点（需登录）
async function saveCheckpoint(courseId: string, state: any) {
  const res = await fetch(`${API_BASE}/api/checkpoint`, {
    method: "POST",
    headers: headers(true),
    body: JSON.stringify({ courseId, state }),
  });
  return handleResponse(res);
  // 返回: { message: "进度已保存", savedAt: "ISO8601" }
}

// 恢复进度断点（需登录）
async function loadCheckpoint(courseId: string) {
  const res = await fetch(`${API_BASE}/api/checkpoint?courseId=${courseId}`, {
    headers: headers(true),
  });
  return handleResponse(res);
  // 返回: { state: FarmProgress | null, savedAt: "ISO8601" }
}

// 使用示例：页面加载时恢复进度
async function onPageLoad(courseId: string) {
  const { state } = await loadCheckpoint(courseId);
  if (state) {
    // 将 state 恢复到前端状态机
    return state;
  }
  // 无断点，从初始状态开始
  return null;
}

// 每次操作后保存断点
async function onUserAction(userId: string, courseId: string, action: any) {
  // 1. 提交进度到后端
  const result = await applyProgressAction(userId, courseId, action);
  // 2. 保存断点（用于异常恢复）
  await saveCheckpoint(courseId, result.progress);
  return result;
}
```

**断点 state 的 FarmProgress 结构与进度同步接口一致。**

## 完整联调脚本示例

以下是一个完整的用户注册 → 学习 → 绑定钱包 → 查询等级的全流程脚本，可直接在浏览器 DevTools Console 中执行：

```javascript
(async () => {
  const API = "http://localhost:3000";

  // 1. 注册
  const regRes = await fetch(`${API}/api/auth`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ username: "dev_test_" + Date.now() }),
  });
  const { user, token } = await regRes.json();
  console.log("注册成功:", user.id, user.username);

  // 2. 获取课程
  const courseRes = await fetch(`${API}/api/courses?courseId=web3`);
  const { course } = await courseRes.json();
  console.log("课程:", course.title, "章节数:", course.chapters.length);

  // 3. 开始学习
  const startRes = await fetch(`${API}/api/progress`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      userId: user.id,
      courseId: "web3",
      action: { type: "start-course" },
    }),
  });
  const { progress: p1, changed } = await startRes.json();
  console.log("开始课程:", p1, "changed:", changed);

  // 4. 完成章节阅读
  for (let i = 0; i < 3; i++) {
    await fetch(`${API}/api/progress`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        userId: user.id,
        courseId: "web3",
        action: { type: "complete-theory", chapter: i },
      }),
    });
  }

  // 5. 完成挑战
  for (let i = 0; i < 3; i++) {
    await fetch(`${API}/api/progress`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        userId: user.id,
        courseId: "web3",
        action: { type: "complete-challenge", chapter: i },
      }),
    });
  }

  // 6. 验证通过
  await fetch(`${API}/api/progress`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      userId: user.id,
      courseId: "web3",
      action: { type: "pass-verification" },
    }),
  });

  // 7. 收获
  await fetch(`${API}/api/progress`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      userId: user.id,
      courseId: "web3",
      action: { type: "harvest" },
    }),
  });

  // 8. 查询等级
  const levelRes = await fetch(`${API}/api/level`, {
    headers: { Authorization: `Bearer ${token}` },
  });
  const { level } = await levelRes.json();
  console.log("等级:", level.current, level.name, "EXP:", level.exp);
})();
```
